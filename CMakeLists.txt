cmake_minimum_required(VERSION 3.21)
project(CodexProxy VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Codex-Proxy currently requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
  _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A00000C)

add_library(minhook STATIC src/buffer.c src/hook.c src/trampoline.c src/hde/hde64.c)
target_include_directories(minhook PUBLIC include PRIVATE src/hde)
add_library(codex_proxy SHARED src/main.cpp src/hooks/Hooks.cpp)
target_include_directories(codex_proxy PRIVATE src include)
target_link_libraries(codex_proxy PRIVATE minhook ws2_32 winhttp iphlpapi)
add_library(version SHARED src/proxy/VersionProxy.cpp src/proxy/ShimEntry.cpp src/proxy/version.def)
target_link_libraries(version PRIVATE codex_proxy)
add_library(tdh SHARED src/proxy/TdhShim.cpp src/proxy/ShimEntry.cpp src/proxy/tdh.def)
target_link_libraries(tdh PRIVATE codex_proxy)
add_executable(codex-proxy src/launcher/CodexCliLauncher.cpp)
target_include_directories(codex-proxy PRIVATE src include)
target_link_libraries(codex-proxy PRIVATE ws2_32 shell32)
add_executable(CodexDesktopLauncher WIN32 src/launcher/CodexDesktopLauncher.cpp)
target_include_directories(CodexDesktopLauncher PRIVATE src include)
target_link_libraries(CodexDesktopLauncher PRIVATE ws2_32 shell32 comdlg32 user32)
add_dependencies(codex-proxy codex_proxy)
add_dependencies(CodexDesktopLauncher codex_proxy version)
install(TARGETS codex-proxy CodexDesktopLauncher RUNTIME DESTINATION bin)
install(TARGETS codex_proxy version tdh RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES LICENSE.txt AUTHORS.txt README.md UPSTREAM.txt config.example.json DESTINATION bin)

option(BUILD_TESTS "Build core and Windows integration tests" OFF)
if(BUILD_TESTS)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  function(core_test name)
    add_executable(${name} tests/${name}.cpp)
    target_include_directories(${name} PRIVATE src include)
    target_link_libraries(${name} PRIVATE ws2_32 iphlpapi)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 45)
  endfunction()
  foreach(name test_ipv6_parser test_udp_policy test_udp_associate_backoff
      test_socket_family test_local_target_policy test_proxy_auth
      test_config test_process_name test_routing test_fake_ip test_protocols test_logger test_launcher_logic)
    core_test(${name})
  endforeach()
  add_executable(test_tdh_shim tests/test_tdh_shim.cpp)
  target_link_libraries(test_tdh_shim PRIVATE ws2_32)
  add_dependencies(test_tdh_shim tdh version)
  add_test(NAME test_tdh_shim COMMAND test_tdh_shim "$<TARGET_FILE:tdh>" "$<TARGET_FILE:version>")
  set_tests_properties(test_tdh_shim PROPERTIES TIMEOUT 30)
  add_executable(test_native_tdh tests/test_native_tdh.cpp)
  set_target_properties(test_native_tdh PROPERTIES OUTPUT_NAME codex)
  target_link_libraries(test_native_tdh PRIVATE tdh)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-config.json"
    CONTENT "{\"proxy\":{\"port\":0},\"target_processes\":[\"codex.exe\"]}")
  add_test(NAME native_tdh_import COMMAND test_native_tdh)
  set_tests_properties(native_tdh_import PROPERTIES TIMEOUT 30
    ENVIRONMENT "CODEX_PROXY_CONFIG=${CMAKE_CURRENT_BINARY_DIR}/native-config.json")
  add_executable(test_hooks tests/test_hooks.cpp)
  target_include_directories(test_hooks PRIVATE src include)
  target_link_libraries(test_hooks PRIVATE ws2_32 winhttp iphlpapi)
  target_compile_definitions(test_hooks PRIVATE _WINSOCK_DEPRECATED_NO_WARNINGS)
  add_dependencies(test_hooks codex_proxy)
  foreach(scenario dns winhttp winhttp_off disabled local scope_main scope_main_on scope_renderer injection duplicate tcp iocp iocp_fail iocp_fail_ex udp udp_event udp_callback udp_immediate_callback)
    add_test(NAME hooks_${scenario} COMMAND test_hooks "$<TARGET_FILE:codex_proxy>" ${scenario})
    set_tests_properties(hooks_${scenario} PROPERTIES TIMEOUT 45)
  endforeach()
  add_test(NAME hooks_child_config_request COMMAND test_hooks "$<TARGET_FILE:codex_proxy>" child_config_request)
  set_tests_properties(hooks_child_config_request PROPERTIES TIMEOUT 45)
  add_executable(test_review_regressions tests/test_review_regressions.cpp)
  target_include_directories(test_review_regressions PRIVATE src include)
  target_link_libraries(test_review_regressions PRIVATE minhook ws2_32 winhttp iphlpapi)
  target_compile_definitions(test_review_regressions PRIVATE _WINSOCK_DEPRECATED_NO_WARNINGS)
  foreach(scenario dns_direct connect_reuse udp_release udp_unconnected udp_immediate byname_local)
    add_test(NAME review_${scenario} COMMAND test_review_regressions ${scenario})
    set_tests_properties(review_${scenario} PROPERTIES TIMEOUT 45)
  endforeach()
  add_executable(test_launcher_child tests/test_launcher_child.cpp)
  target_include_directories(test_launcher_child PRIVATE src include)
  target_link_libraries(test_launcher_child PRIVATE ws2_32)
  add_executable(test_launcher_e2e tests/test_launcher_e2e.cpp)
  target_include_directories(test_launcher_e2e PRIVATE src include)
  target_link_libraries(test_launcher_e2e PRIVATE ws2_32)
  add_dependencies(test_launcher_e2e codex-proxy test_launcher_child CodexDesktopLauncher)
  foreach(protocol socks5 http pipes attach failure job)
    add_test(NAME launcher_${protocol} COMMAND test_launcher_e2e "$<TARGET_FILE:codex-proxy>" "$<TARGET_FILE:test_launcher_child>" ${protocol})
    set_tests_properties(launcher_${protocol} PROPERTIES TIMEOUT 45)
  endforeach()
  add_test(NAME desktop_dry_run COMMAND CodexDesktopLauncher --target "$<TARGET_FILE:test_launcher_child>" --dry-run)
  add_test(NAME powershell_ast COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_powershell_syntax.ps1" -Root "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME installer_lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_installer.py" --root "${CMAKE_CURRENT_SOURCE_DIR}" --build-dir "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(installer_lifecycle PROPERTIES TIMEOUT 120)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config_web.py" --check RESULT_VARIABLE BROWSER_QA_AVAILABLE OUTPUT_QUIET ERROR_QUIET)
  if(BROWSER_QA_AVAILABLE EQUAL 0)
    add_test(NAME config_web_browser COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config_web.py" --root "${CMAKE_CURRENT_SOURCE_DIR}" --evidence "${CMAKE_CURRENT_BINARY_DIR}/evidence")
    set_tests_properties(config_web_browser PROPERTIES TIMEOUT 60)
  endif()
endif()
